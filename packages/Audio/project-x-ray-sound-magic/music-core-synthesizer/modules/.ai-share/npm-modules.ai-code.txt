import type {
  DistortionAudioSampleValue,
  DistortionSampleRateHzValue,
  IBitCrushDistortionDefinition,
} from "@monorepo/music-core-synthesizer-distortion-contracts";

import { DistortionMath } from "../math/index.js";
import type { IDistortionAlgorithm } from "./IDistortionAlgorithm.js";

export class BitCrushDistortionAlgorithm
  implements IDistortionAlgorithm<IBitCrushDistortionDefinition> {
  private holdCounter = 0;
  private heldSample: DistortionAudioSampleValue = 0;

  public process(
    input: DistortionAudioSampleValue,
    definition: IBitCrushDistortionDefinition,
    _sampleRateHz: DistortionSampleRateHzValue,
  ): DistortionAudioSampleValue {
    if (this.holdCounter <= 0) {
      this.heldSample = DistortionMath.quantizeBipolar(input, definition.bitDepth);
      this.holdCounter = definition.sampleRateReduction;
    }

    this.holdCounter -= 1;

    return this.heldSample;
  }

  public reset(): void {
    this.holdCounter = 0;
    this.heldSample = 0;
  }
}

import {
  DistortionCurveKind,
  type DistortionAudioSampleValue,
  type DistortionSampleRateHzValue,
  type ISoftClipDistortionDefinition,
} from "@monorepo/music-core-synthesizer-distortion-contracts";

import { DistortionMath } from "../math/index.js";
import type { IDistortionAlgorithm } from "./IDistortionAlgorithm.js";

export class SoftClipDistortionAlgorithm
  implements IDistortionAlgorithm<ISoftClipDistortionDefinition> {
  public process(
    input: DistortionAudioSampleValue,
    definition: ISoftClipDistortionDefinition,
    _sampleRateHz: DistortionSampleRateHzValue,
  ): DistortionAudioSampleValue {
    const linear = DistortionMath.clampBipolar(input);
    const curved = this.evaluateCurve(linear, definition);

    return DistortionMath.mix(linear, curved, definition.amount);
  }

  public reset(): void {}

  private evaluateCurve(
    input: DistortionAudioSampleValue,
    definition: ISoftClipDistortionDefinition,
  ): DistortionAudioSampleValue {
    const drive = 1 + definition.amount * 12;

    switch (definition.curve) {
      case DistortionCurveKind.Linear:
        return input;

      case DistortionCurveKind.Cubic:
        return DistortionMath.clampBipolar(input - (input * input * input) / 3);

      case DistortionCurveKind.Arctangent:
        return DistortionMath.clampBipolar((2 / Math.PI) * Math.atan(drive * input));

      case DistortionCurveKind.Tanh:
        return DistortionMath.clampBipolar(Math.tanh(drive * input));

      default:
        throw new Error(`Unsupported distortion curve kind: ${String(definition.curve)}`);
    }
  }
}

// FILE: src\math\DistortionMath.ts

export class DistortionMath {
  public static clamp(
    value: number,
    minimum: number,
    maximum: number,
  ): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value < minimum) {
      return minimum;
    }

    if (value > maximum) {
      return maximum;
    }

    return value;
  }

  public static clampBipolar(value: number): number {
    return DistortionMath.clamp(value, -1, 1);
  }

  public static mix(
    dry: number,
    wet: number,
    mix: number,
  ): number {
    return DistortionMath.clampBipolar(
      dry * (1 - mix) + wet * mix,
    );
  }

  public static quantizeBipolar(
    value: number,
    bitDepth: number,
  ): number {
    const safeBitDepth = Math.max(1, Math.min(32, Math.trunc(bitDepth)));
    const levelCount = Math.pow(2, safeBitDepth);
    const normalized = (DistortionMath.clampBipolar(value) + 1) / 2;
    const quantized = Math.round(normalized * (levelCount - 1)) / (levelCount - 1);

    return DistortionMath.clampBipolar(quantized * 2 - 1);
  }

  public static foldback(
    value: number,
    threshold: number,
  ): number {
    const safeThreshold = DistortionMath.clamp(threshold, 0.000_001, 1);
    let folded = value;

    while (folded > safeThreshold || folded < -safeThreshold) {
      if (folded > safeThreshold) {
        folded = safeThreshold - (folded - safeThreshold);
        continue;
      }

      if (folded < -safeThreshold) {
        folded = -safeThreshold - (folded + safeThreshold);
      }
    }

    return DistortionMath.clampBipolar(folded);
  }
}

import {
  DistortionAudioSampleSpecification,
  DistortionDefinitionValidator,
  DistortionKind,
  DistortionSampleRateHzSpecification,
  type DistortionAudioSampleValue,
  type DistortionDefinition,
  type IDistortion,
  type IDistortionProcessRequest,
  type IDistortionProcessResult,
  type IDistortionStateSnapshot,
} from "@monorepo/music-core-synthesizer-distortion-contracts";

import type { IDistortionAlgorithm } from "../algorithms/index.js";
import { DistortionMath } from "../math/index.js";

export class Distortion implements IDistortion {
  private lastInput: DistortionAudioSampleValue = 0;
  private lastOutput: DistortionAudioSampleValue = 0;

  private readonly audioSampleSpecification =
    DistortionAudioSampleSpecification.minusOneToOne();

  private readonly sampleRateSpecification =
    DistortionSampleRateHzSpecification.positiveFinite();

  public constructor(
    public readonly definition: DistortionDefinition,
    private readonly algorithm: IDistortionAlgorithm,
    private readonly definitionValidator: DistortionDefinitionValidator,
  ) {
    this.definitionValidator.validate(definition);
  }

  public process(request: IDistortionProcessRequest): IDistortionProcessResult {
    this.assertInput(request.input);
    this.assertSampleRate(request.sampleRateHz);

    this.lastInput = request.input;

    const dry = request.input;
    const driven = DistortionMath.clampBipolar(request.input * this.definition.inputGain);
    const wetRaw = this.algorithm.process(driven, this.definition, request.sampleRateHz);
    const wet = DistortionMath.clampBipolar(wetRaw * this.definition.outputGain);
    const output = DistortionMath.mix(dry, wet, this.definition.mix);

    this.lastOutput = output;

    return {
      input: request.input,
      output,
      wet,
      dry,
    };
  }

  public snapshot(): IDistortionStateSnapshot {
    return {
      definition: this.definition,
      lastInput: this.lastInput,
      lastOutput: this.lastOutput,
      isBypassed: this.definition.kind === DistortionKind.Bypass,
    };
  }

  public reset(): void {
    this.lastInput = 0;
    this.lastOutput = 0;
    this.algorithm.reset();
  }

  private assertInput(input: DistortionAudioSampleValue): void {
    if (!this.audioSampleSpecification.isSatisfiedBy(input)) {
      throw new Error(`input must be between -1 and 1. Actual value: ${String(input)}.`);
    }
  }

  private assertSampleRate(sampleRateHz: number): void {
    if (!this.sampleRateSpecification.isSatisfiedBy(sampleRateHz)) {
      throw new Error(`sampleRateHz must be finite and greater than 0. Actual value: ${String(sampleRateHz)}.`);
    }
  }
}

      && this.isBaseValid(candidate)
      && BitDepthSpecification.standardRange().isSatisfiedBy(candidate.bitDepth)
      && SampleRateReductionSpecification.standardRange().isSatisfiedBy(candidate.sampleRateReduction);
  }
}

export class DistortionDefinitionSpecification
  extends Specification<DistortionDefinition> {
  public isSatisfiedBy(candidate: DistortionDefinition): boolean {
    switch (candidate.kind) {
      case DistortionKind.Bypass:
        return new BypassDistortionDefinitionSpecification().isSatisfiedBy(candidate);

      case DistortionKind.HardClip:
        return new HardClipDistortionDefinitionSpecification().isSatisfiedBy(candidate);

      case DistortionKind.SoftClip:
        return new SoftClipDistortionDefinitionSpecification().isSatisfiedBy(candidate);

      case DistortionKind.Tanh:
        return new TanhDistortionDefinitionSpecification().isSatisfiedBy(candidate);

      case DistortionKind.Foldback:
        return new FoldbackDistortionDefinitionSpecification().isSatisfiedBy(candidate);

      case DistortionKind.BitCrush:
        return new BitCrushDistortionDefinitionSpecification().isSatisfiedBy(candidate);

      default:
        return false;
    }
  }

  public static valid(): ISpecification<DistortionDefinition> {
    return new DistortionDefinitionSpecification();
  }
}

import {
  DistortionCurveKind,
  DistortionKind,
  type DistortionDefinition,
  type IBitCrushDistortionDefinition,
  type IFoldbackDistortionDefinition,
  type IHardClipDistortionDefinition,
  type ISoftClipDistortionDefinition,
  type ITanhDistortionDefinition,
} from "@monorepo/music-core-synthesizer-distortion-contracts";

import { WebAudioDistortionMath } from "../math/index.js";
import type { IWaveShaperCurveFactory } from "./IWaveShaperCurveFactory.js";

export interface IWaveShaperCurveFactoryOptions {
  readonly curveLength: number;
}

export class WaveShaperCurveFactory implements IWaveShaperCurveFactory {
  private readonly curveLength: number;

  public constructor(options: IWaveShaperCurveFactoryOptions = { curveLength: 2048 }) {
    if (!Number.isInteger(options.curveLength) || options.curveLength < 2) {
      throw new Error("curveLength must be an integer greater than or equal to 2.");
    }

    this.curveLength = options.curveLength;
  }

  public create(definition: DistortionDefinition): Float32Array {
    switch (definition.kind) {
      case DistortionKind.Bypass:
        return this.createCurve((input) => input);

      case DistortionKind.HardClip:
        return this.createHardClipCurve(definition);

      case DistortionKind.SoftClip:
        return this.createSoftClipCurve(definition);

      case DistortionKind.Tanh:
        return this.createTanhCurve(definition);

      case DistortionKind.Foldback:
        return this.createFoldbackCurve(definition);

      case DistortionKind.BitCrush:
        return this.createBitCrushCurve(definition);
    }
  }

  private createHardClipCurve(definition: IHardClipDistortionDefinition): Float32Array {
    return this.createCurve((input) =>
      WebAudioDistortionMath.clamp(
        input,
        -definition.threshold,
        definition.threshold,
      ),
    );
  }

  private createSoftClipCurve(definition: ISoftClipDistortionDefinition): Float32Array {
    const drive = 1 + definition.amount * 12;

    return this.createCurve((input) => {
      switch (definition.curve) {
        case DistortionCurveKind.Linear:
          return input;

        case DistortionCurveKind.Cubic:
          return WebAudioDistortionMath.clampBipolar(
            input - (input * input * input) / 3,
          );

        case DistortionCurveKind.Arctangent:
          return WebAudioDistortionMath.clampBipolar(
            (2 / Math.PI) * Math.atan(drive * input),
          );

        case DistortionCurveKind.Tanh:
          return WebAudioDistortionMath.clampBipolar(
            Math.tanh(drive * input),
          );

        default:
          throw new Error(`Unsupported distortion curve kind: ${String(definition.curve)}`);
      }
    });
  }

  private createTanhCurve(definition: ITanhDistortionDefinition): Float32Array {
    const drive = 1 + definition.amount * 16;

    return this.createCurve((input) =>
      WebAudioDistortionMath.clampBipolar(Math.tanh(input * drive)),
    );
  }

  private createFoldbackCurve(definition: IFoldbackDistortionDefinition): Float32Array {
    return this.createCurve((input) =>
      WebAudioDistortionMath.foldback(input, definition.threshold),
    );
  }

  private createBitCrushCurve(definition: IBitCrushDistortionDefinition): Float32Array {
    if (definition.sampleRateReduction !== 1) {
      throw new Error(
        "Bit-crush sampleRateReduction requires an AudioWorklet adapter. WaveShaperNode can only approximate bit-depth quantization.",
      );
    }

    return this.createCurve((input) =>
      WebAudioDistortionMath.quantizeBipolar(input, definition.bitDepth),
    );
  }

  private createCurve(transform: (input: number) => number): Float32Array {
    const curve = new Float32Array(this.curveLength);

    for (let index = 0; index < this.curveLength; index += 1) {
      const input = (index / (this.curveLength - 1)) * 2 - 1;
      curve[index] = WebAudioDistortionMath.clampBipolar(transform(input));
    }

    return curve;
  }
}

    this.wetInputGainNode.connect(this.waveShaperNode);
    this.waveShaperNode.connect(this.wetOutputGainNode);
    this.wetOutputGainNode.connect(this.outputNode);

    this.definitionValidator.validate(definition);

    this.currentDefinition = definition;
    this.waveShaperNode.curve = this.curveFactory.create(definition);

    if (definition.kind === DistortionKind.Bypass) {
      this.dryGainNode.gain.value = 1;
      this.wetInputGainNode.gain.value = 1;

export class WebAudioDistortionMath {
  public static clamp(
    value: number,
    minimum: number,
    maximum: number,
  ): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value < minimum) {
      return minimum;
    }

    if (value > maximum) {
      return maximum;
    }

    return value;
  }

  public static clampBipolar(value: number): number {
    return WebAudioDistortionMath.clamp(value, -1, 1);
  }

  public static foldback(
    value: number,
    threshold: number,
  ): number {
    const safeThreshold = WebAudioDistortionMath.clamp(threshold, 0.000_001, 1);
    let folded = value;

    while (folded > safeThreshold || folded < -safeThreshold) {
      if (folded > safeThreshold) {
        folded = safeThreshold - (folded - safeThreshold);
        continue;
      }

      folded = -safeThreshold - (folded + safeThreshold);
    }

    return WebAudioDistortionMath.clampBipolar(folded);
  }

  public static quantizeBipolar(
    value: number,
    bitDepth: number,
  ): number {
    const safeBitDepth = Math.max(1, Math.min(32, Math.trunc(bitDepth)));
    const levelCount = Math.pow(2, safeBitDepth);
    const normalized = (WebAudioDistortionMath.clampBipolar(value) + 1) / 2;
    const quantized = Math.round(normalized * (levelCount - 1)) / (levelCount - 1);

    return WebAudioDistortionMath.clampBipolar(quantized * 2 - 1);
  }
}

export class EnvelopeMath {
  public static clamp01(value: number): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value < 0) {
      return 0;
    }

    if (value > 1) {
      return 1;
    }

    return value;
  }

  public static progress(elapsedSeconds: number, durationSeconds: number): number {
    if (durationSeconds <= 0) {
      return 1;
    }

    return EnvelopeMath.clamp01(elapsedSeconds / durationSeconds);
  }

  public static interpolate(
    startLevel: number,
    endLevel: number,
    progress: number,
  ): number {
    return startLevel + (endLevel - startLevel) * EnvelopeMath.clamp01(progress);
  }
}

export class LinearEnvelopeCurveEvaluator implements IEnvelopeCurveEvaluator {
  public evaluate(progress: number): number {
    return EnvelopeMath.clamp01(progress);
  }
}

export class ExponentialEnvelopeCurveEvaluator implements IEnvelopeCurveEvaluator {
  public evaluate(progress: number): number {
    const value = EnvelopeMath.clamp01(progress);

    return value * value;
  }
}

export class LogarithmicEnvelopeCurveEvaluator implements IEnvelopeCurveEvaluator {
  public evaluate(progress: number): number {
    return Math.sqrt(EnvelopeMath.clamp01(progress));
  }
}

export class InstantEnvelopeCurveEvaluator implements IEnvelopeCurveEvaluator {
  public evaluate(progress: number): number {
    return EnvelopeMath.clamp01(progress) <= 0 ? 0 : 1;
  }
}

      case EnvelopeCurveKind.Exponential:
        return new ExponentialEnvelopeCurveEvaluator();

      case EnvelopeCurveKind.Logarithmic:

interface IEnvelopeEvaluation {
  readonly level: EnvelopeLevelValue;
  readonly stage: typeof EnvelopeStage[keyof typeof EnvelopeStage];
  readonly isComplete: boolean;
}

export class Envelope implements IEnvelope {
  private startedAtSeconds: EnvelopeSecondsValue | null = null;
  private releasedAtSeconds: EnvelopeSecondsValue | null = null;
  private releaseStartLevel: EnvelopeLevelValue = 0;
  private currentLevel: EnvelopeLevelValue = 0;
  private currentStage: typeof EnvelopeStage[keyof typeof EnvelopeStage] = EnvelopeStage.Idle;
  private isGateOpen = false;
  private isEnvelopeComplete = false;
  private velocity: EnvelopeVelocityValue = 1;

  private readonly secondsSpecification = new EnvelopeSecondsSpecification();
  private readonly velocitySpecification = new EnvelopeVelocitySpecification();

  public constructor(
    public readonly definition: EnvelopeDefinition,
    private readonly curveEvaluatorFactory: IEnvelopeCurveEvaluatorFactory,
  ) {}

  public gateOn(request: IEnvelopeGateOnRequest): void {
    this.assertSeconds(request.timeSeconds);
    this.assertVelocity(request.velocity);

    this.startedAtSeconds = request.timeSeconds;
    this.releasedAtSeconds = null;
    this.releaseStartLevel = 0;
    this.currentLevel = 0;
    this.currentStage = EnvelopeStage.Attack;
    this.isGateOpen = true;
    this.isEnvelopeComplete = false;
    this.velocity = request.velocity;
  }

  public gateOff(request: IEnvelopeGateOffRequest): void {
    this.assertSeconds(request.timeSeconds);

    if (this.startedAtSeconds === null) {
      return;
    }

    if (this.definition.kind === EnvelopeKind.OneShot) {
      return;
    }

    const evaluation = this.evaluateAt(request.timeSeconds);

    this.releasedAtSeconds = request.timeSeconds;
    this.releaseStartLevel = evaluation.level;
    this.currentLevel = evaluation.level;
    this.currentStage = EnvelopeStage.Release;
    this.isGateOpen = false;
    this.isEnvelopeComplete = false;
  }

  public sample(request: IEnvelopeSampleRequest): IEnvelopeSample {
    this.assertSeconds(request.timeSeconds);

    const evaluation = this.evaluateAt(request.timeSeconds);

    this.currentLevel = evaluation.level;
    this.currentStage = evaluation.stage;
    this.isEnvelopeComplete = evaluation.isComplete;

    if (evaluation.isComplete) {
      this.isGateOpen = false;
    }

    return {
      timeSeconds: request.timeSeconds,
      level: evaluation.level,
      stage: evaluation.stage,
      isComplete: evaluation.isComplete,
    };
  }

  public snapshot(): IEnvelopeStateSnapshot {
    return {
      stage: this.currentStage,
      startedAtSeconds: this.startedAtSeconds,
      releasedAtSeconds: this.releasedAtSeconds,
      releaseStartLevel: this.releaseStartLevel,
      currentLevel: this.currentLevel,
      isGateOpen: this.isGateOpen,
      isComplete: this.isEnvelopeComplete,
    };
  }

  public reset(): void {
    this.startedAtSeconds = null;
    this.releasedAtSeconds = null;
    this.releaseStartLevel = 0;
    this.currentLevel = 0;
    this.currentStage = EnvelopeStage.Idle;
    this.isGateOpen = false;
    this.isEnvelopeComplete = false;
    this.velocity = 1;
  }

  private evaluateAt(timeSeconds: EnvelopeSecondsValue): IEnvelopeEvaluation {
    if (this.startedAtSeconds === null) {
      return {
        level: 0,
        stage: EnvelopeStage.Idle,
        isComplete: false,
      };
    }

    if (this.releasedAtSeconds !== null && this.definition.kind !== EnvelopeKind.OneShot) {
      return this.evaluateRelease(timeSeconds);
    }

    switch (this.definition.kind) {
      case EnvelopeKind.Adsr:
        return this.evaluateAdsr(this.definition, timeSeconds);

      case EnvelopeKind.Ahdsr:
        return this.evaluateAhdsr(this.definition, timeSeconds);

      case EnvelopeKind.Ar:
        return this.evaluateAr(this.definition, timeSeconds);

      case EnvelopeKind.Asr:
        return this.evaluateAsr(this.definition, timeSeconds);

      case EnvelopeKind.OneShot:
        return this.evaluateOneShot(this.definition, timeSeconds);
    }
  }

  private evaluateAdsr(
    definition: IAdsrEnvelopeDefinition,
    timeSeconds: EnvelopeSecondsValue,
  ): IEnvelopeEvaluation {
    return this.evaluateAttackDecaySustain(
      timeSeconds,
      definition.attackSeconds,
      0,
      definition.decaySeconds,
      definition.sustainLevel,
      definition.attackCurve,
      definition.decayCurve,
    );
  }

  private evaluateAhdsr(
    definition: IAhdsrEnvelopeDefinition,
    timeSeconds: EnvelopeSecondsValue,
  ): IEnvelopeEvaluation {
    return this.evaluateAttackDecaySustain(
      timeSeconds,
      definition.attackSeconds,
      definition.holdSeconds,
      definition.decaySeconds,
      definition.sustainLevel,
      definition.attackCurve,
      definition.decayCurve,
    );
  }

  private evaluateAr(
    definition: IArEnvelopeDefinition,
    timeSeconds: EnvelopeSecondsValue,
  ): IEnvelopeEvaluation {
    const elapsedSeconds = this.getElapsedSinceStart(timeSeconds);
    const peakLevel = this.velocity;

    if (definition.attackSeconds > 0 && elapsedSeconds < definition.attackSeconds) {
      return {
        level: this.evaluateCurveLevel(
          0,
          peakLevel,
          elapsedSeconds,
          definition.attackSeconds,
          definition.attackCurve,
        ),
        stage: EnvelopeStage.Attack,
        isComplete: false,
      };
    }

    return {
      level: peakLevel,
      stage: EnvelopeStage.Sustain,
      isComplete: false,
    };
  }

  private evaluateAsr(
    definition: IAsrEnvelopeDefinition,
    timeSeconds: EnvelopeSecondsValue,
  ): IEnvelopeEvaluation {
    const elapsedSeconds = this.getElapsedSinceStart(timeSeconds);
    const peakLevel = this.velocity;
    const sustainLevel = this.velocity * definition.sustainLevel;

    if (definition.attackSeconds > 0 && elapsedSeconds < definition.attackSeconds) {
      return {
        level: this.evaluateCurveLevel(
          0,
          peakLevel,
          elapsedSeconds,
          definition.attackSeconds,
          definition.attackCurve,
        ),
        stage: EnvelopeStage.Attack,
        isComplete: false,
      };
    }

    return {
      level: sustainLevel,
      stage: EnvelopeStage.Sustain,
      isComplete: false,
    };
  }

  private evaluateOneShot(
    definition: IOneShotEnvelopeDefinition,
    timeSeconds: EnvelopeSecondsValue,
  ): IEnvelopeEvaluation {
    let elapsedSeconds = this.getElapsedSinceStart(timeSeconds);
    const peakLevel = this.velocity;

    if (definition.attackSeconds > 0 && elapsedSeconds < definition.attackSeconds) {
      return {
        level: this.evaluateCurveLevel(
          0,
          peakLevel,
          elapsedSeconds,
          definition.attackSeconds,
          definition.attackCurve,
        ),
        stage: EnvelopeStage.Attack,
        isComplete: false,
      };
    }

    elapsedSeconds -= definition.attackSeconds;

    if (definition.decaySeconds > 0 && elapsedSeconds < definition.decaySeconds) {
      return {
        level: this.evaluateCurveLevel(
          peakLevel,
          0,
          elapsedSeconds,
          definition.decaySeconds,
          definition.decayCurve,
        ),
        stage: EnvelopeStage.Decay,
        isComplete: false,
      };
    }

    return {
      level: 0,
      stage: EnvelopeStage.Complete,
      isComplete: true,
    };
  }

  private evaluateAttackDecaySustain(
    timeSeconds: EnvelopeSecondsValue,
    attackSeconds: EnvelopeSecondsValue,
    holdSeconds: EnvelopeSecondsValue,
    decaySeconds: EnvelopeSecondsValue,
    sustainLevel: EnvelopeLevelValue,
    attackCurve: IEnvelopeCurveDefinition,
    decayCurve: IEnvelopeCurveDefinition,
  ): IEnvelopeEvaluation {
    let elapsedSeconds = this.getElapsedSinceStart(timeSeconds);
    const peakLevel = this.velocity;
    const targetSustainLevel = this.velocity * sustainLevel;

    if (attackSeconds > 0 && elapsedSeconds < attackSeconds) {
      return {
        level: this.evaluateCurveLevel(
          0,
          peakLevel,
          elapsedSeconds,
          attackSeconds,
          attackCurve,
        ),
        stage: EnvelopeStage.Attack,
        isComplete: false,
      };
    }

    elapsedSeconds -= attackSeconds;

    if (holdSeconds > 0 && elapsedSeconds < holdSeconds) {
      return {
        level: peakLevel,
        stage: EnvelopeStage.Hold,
        isComplete: false,
      };
    }

    elapsedSeconds -= holdSeconds;

    if (decaySeconds > 0 && elapsedSeconds < decaySeconds) {
      return {
        level: this.evaluateCurveLevel(
          peakLevel,
          targetSustainLevel,
          elapsedSeconds,
          decaySeconds,
          decayCurve,
        ),
        stage: EnvelopeStage.Decay,
        isComplete: false,
      };
    }

    return {
      level: targetSustainLevel,
      stage: EnvelopeStage.Sustain,
      isComplete: false,
    };
  }

  private evaluateRelease(timeSeconds: EnvelopeSecondsValue): IEnvelopeEvaluation {
    if (this.releasedAtSeconds === null) {
      throw new Error("Cannot evaluate release before release has started.");
    }

    const releaseSeconds = this.getReleaseSeconds(this.definition);
    const releaseCurve = this.getReleaseCurve(this.definition);

    if (releaseSeconds <= 0) {
      return {
        level: 0,
        stage: EnvelopeStage.Complete,
        isComplete: true,
      };
    }

    const elapsedSeconds = Math.max(0, timeSeconds - this.releasedAtSeconds);

    if (elapsedSeconds >= releaseSeconds) {
      return {
        level: 0,
        stage: EnvelopeStage.Complete,
        isComplete: true,
      };
    }

    return {
      level: this.evaluateCurveLevel(
        this.releaseStartLevel,
        0,
        elapsedSeconds,
        releaseSeconds,
        releaseCurve,
      ),
      stage: EnvelopeStage.Release,
      isComplete: false,
    };
  }

  private evaluateCurveLevel(
    startLevel: EnvelopeLevelValue,
    endLevel: EnvelopeLevelValue,
    elapsedSeconds: EnvelopeSecondsValue,
    durationSeconds: EnvelopeSecondsValue,
    curve: IEnvelopeCurveDefinition,
  ): EnvelopeLevelValue {
    const evaluator = this.curveEvaluatorFactory.create(curve);
    const progress = EnvelopeMath.progress(elapsedSeconds, durationSeconds);

    return EnvelopeMath.interpolate(startLevel, endLevel, evaluator.evaluate(progress));
  }

  private getElapsedSinceStart(timeSeconds: EnvelopeSecondsValue): EnvelopeSecondsValue {
    if (this.startedAtSeconds === null) {
      return 0;
    }

    return Math.max(0, timeSeconds - this.startedAtSeconds);
  }

  private getReleaseSeconds(definition: EnvelopeDefinition): EnvelopeSecondsValue {
    switch (definition.kind) {
      case EnvelopeKind.Adsr:
      case EnvelopeKind.Ahdsr:
      case EnvelopeKind.Ar:
      case EnvelopeKind.Asr:
        return definition.releaseSeconds;

      case EnvelopeKind.OneShot:
        return 0;
    }
  }

  private getReleaseCurve(definition: EnvelopeDefinition): IEnvelopeCurveDefinition {
    switch (definition.kind) {
      case EnvelopeKind.Adsr:
      case EnvelopeKind.Ahdsr:
      case EnvelopeKind.Ar:
      case EnvelopeKind.Asr:
        return definition.releaseCurve;

      case EnvelopeKind.OneShot:
        return {
          kind: EnvelopeCurveKind.Instant,
        };
    }
  }

  private assertSeconds(value: EnvelopeSecondsValue): void {
    if (!this.secondsSpecification.isSatisfiedBy(value)) {
      throw new Error(`timeSeconds must be finite and greater than or equal to 0. Actual value: ${String(value)}.`);
    }
  }

  private assertVelocity(value: EnvelopeVelocityValue): void {
    if (!this.velocitySpecification.isSatisfiedBy(value)) {
      throw new Error(`velocity must be between 0 and 1. Actual value: ${String(value)}.`);
    }
  }
}

export class EnvelopeFactory implements IEnvelopeFactory {
  public constructor(
    private readonly definitionValidator: EnvelopeDefinitionValidator,
    private readonly curveEvaluatorFactory: IEnvelopeCurveEvaluatorFactory,
  ) {}

  public static createDefault(): EnvelopeFactory {
    return new EnvelopeFactory(
      EnvelopeDefinitionValidator.createDefault(),
      new EnvelopeCurveEvaluatorFactory(),
    );
  }

  public create(definition: EnvelopeDefinition): IEnvelope {
    this.definitionValidator.validate(definition);

    return new Envelope(definition, this.curveEvaluatorFactory);
  }
}

import { WebAudioEnvelopeMath } from "../math/index.js";
import type { IWebAudioAudioParamLike } from "../ports/index.js";
import type { IEnvelopeAutomationScheduler } from "./IEnvelopeAutomationScheduler.js";

export class EnvelopeAutomationScheduler implements IEnvelopeAutomationScheduler {
  private readonly secondsSpecification = new EnvelopeSecondsSpecification();
  private readonly velocitySpecification = new EnvelopeVelocitySpecification();

  public constructor(
    private readonly definitionValidator: EnvelopeDefinitionValidator,
  ) {}

  public static createDefault(): EnvelopeAutomationScheduler {
    return new EnvelopeAutomationScheduler(
      EnvelopeDefinitionValidator.createDefault(),
    );
  }

  public scheduleGateOn(
    definition: EnvelopeDefinition,
    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOnRequest,
  ): void {
    this.definitionValidator.validate(definition);
    this.assertSeconds(request.timeSeconds);
    this.assertVelocity(request.velocity);

    parameter.cancelScheduledValues(request.timeSeconds);
    parameter.setValueAtTime(0, request.timeSeconds);

    switch (definition.kind) {
      case EnvelopeKind.Adsr:
        this.scheduleAdsr(definition, parameter, request);
        return;

      case EnvelopeKind.Ahdsr:
        this.scheduleAhdsr(definition, parameter, request);
        return;

      case EnvelopeKind.Ar:
        this.scheduleAr(definition, parameter, request);
        return;

      case EnvelopeKind.Asr:
        this.scheduleAsr(definition, parameter, request);
        return;

      case EnvelopeKind.OneShot:
        this.scheduleOneShot(definition, parameter, request);
        return;
    }
  }

  public scheduleGateOff(
    definition: EnvelopeDefinition,
    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOffRequest,
  ): void {
    this.definitionValidator.validate(definition);
    this.assertSeconds(request.timeSeconds);

    if (definition.kind === EnvelopeKind.OneShot) {
      return;
    }

    const releaseSeconds = this.getReleaseSeconds(definition);
    const releaseCurve = this.getReleaseCurve(definition);
    const releaseStartLevel = WebAudioEnvelopeMath.levelAtRelease(parameter.value);

    parameter.cancelScheduledValues(request.timeSeconds);
    parameter.setValueAtTime(releaseStartLevel, request.timeSeconds);

    this.scheduleRamp(
      parameter,
      releaseStartLevel,
      0,
      request.timeSeconds,
      releaseSeconds,
      releaseCurve,
    );
  }

  public reset(
    parameter: IWebAudioAudioParamLike,
    timeSeconds: number,
  ): void {
    this.assertSeconds(timeSeconds);

    parameter.cancelScheduledValues(timeSeconds);
    parameter.setValueAtTime(0, timeSeconds);
  }

  private scheduleAdsr(
    definition: IAdsrEnvelopeDefinition,
    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOnRequest,
  ): void {
    this.scheduleAttackDecaySustain(
      parameter,
      request.timeSeconds,
      request.velocity,
      definition.attackSeconds,
      0,
      definition.decaySeconds,
      definition.sustainLevel,
      definition.attackCurve,
      definition.decayCurve,
    );

  private scheduleAhdsr(
    definition: IAhdsrEnvelopeDefinition,
    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOnRequest,
  ): void {
    this.scheduleAttackDecaySustain(
      parameter,
      request.timeSeconds,
      request.velocity,
      definition.attackSeconds,
      definition.holdSeconds,
      definition.decaySeconds,
      definition.sustainLevel,
      definition.attackCurve,
      definition.decayCurve,
    );
  }

  private scheduleAr(
    definition: IArEnvelopeDefinition,
    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOnRequest,
  ): void {
    this.scheduleRamp(
      parameter,
      0,
      request.velocity,
      request.timeSeconds,
      definition.attackSeconds,
      definition.attackCurve,

    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOnRequest,

      parameter,
      0,
      request.velocity * definition.sustainLevel,
      request.timeSeconds,
      definition.attackSeconds,

    );
  }

  private scheduleOneShot(
    definition: IOneShotEnvelopeDefinition,
    parameter: IWebAudioAudioParamLike,
    request: IEnvelopeGateOnRequest,
  ): void {
    const attackEndTime = request.timeSeconds + definition.attackSeconds;

    this.scheduleRamp(
      parameter,
      0,
      request.velocity,
      request.timeSeconds,
      definition.attackSeconds,
      definition.attackCurve,
    );

    this.scheduleRamp(
      parameter,
      request.velocity,
      0,
      attackEndTime,
      definition.decaySeconds,
      definition.decayCurve,
    );
  }

  private scheduleAttackDecaySustain(
    parameter: IWebAudioAudioParamLike,
    startTimeSeconds: EnvelopeSecondsValue,
    velocity: EnvelopeLevelValue,
    attackSeconds: EnvelopeSecondsValue,
    holdSeconds: EnvelopeSecondsValue,
    decaySeconds: EnvelopeSecondsValue,
    sustainLevel: EnvelopeLevelValue,
    attackCurve: IEnvelopeCurveDefinition,
    decayCurve: IEnvelopeCurveDefinition,
  ): void {
    const attackEndTime = startTimeSeconds + attackSeconds;
    const holdEndTime = attackEndTime + holdSeconds;
    const decayEndTime = holdEndTime + decaySeconds;
    const targetSustainLevel = velocity * sustainLevel;

    this.scheduleRamp(
      parameter,
      0,
      velocity,
      startTimeSeconds,
      attackSeconds,
      attackCurve,
    );

    if (holdSeconds > 0) {
      parameter.setValueAtTime(velocity, holdEndTime);
    }

    this.scheduleRamp(
      parameter,
      velocity,
      targetSustainLevel,
      holdEndTime,
      decaySeconds,
      decayCurve,
    );

    parameter.setValueAtTime(targetSustainLevel, decayEndTime);
  }

private scheduleRamp(
  parameter: IWebAudioAudioParamLike,
  startLevel: EnvelopeLevelValue,
  endLevel: EnvelopeLevelValue,
  startTimeSeconds: EnvelopeSecondsValue,
  durationSeconds: EnvelopeSecondsValue,
  curve: IEnvelopeCurveDefinition,
): void {
  const endTimeSeconds = startTimeSeconds + durationSeconds;

  parameter.setValueAtTime(startLevel, startTimeSeconds);

  if (durationSeconds <= 0) {
    parameter.setValueAtTime(endLevel, startTimeSeconds);
    parameter.value = endLevel;
    return;
  }

  switch (curve.kind) {
    case EnvelopeCurveKind.Linear:
      parameter.linearRampToValueAtTime(endLevel, endTimeSeconds);
      parameter.value = endLevel;
      return;

    case EnvelopeCurveKind.Exponential:
    case EnvelopeCurveKind.Logarithmic:
      parameter.setTargetAtTime(
        endLevel,
        startTimeSeconds,
        WebAudioEnvelopeMath.safeTimeConstant(durationSeconds),
      );
      parameter.setValueAtTime(endLevel, endTimeSeconds);
      parameter.value = endLevel;
      return;

    case EnvelopeCurveKind.Instant:
      parameter.setValueAtTime(endLevel, startTimeSeconds);
      parameter.value = endLevel;
      return;
  }
}

  private getReleaseSeconds(definition: EnvelopeDefinition): EnvelopeSecondsValue {
    switch (definition.kind) {
      case EnvelopeKind.Adsr:
      case EnvelopeKind.Ahdsr:
      case EnvelopeKind.Ar:
      case EnvelopeKind.Asr:
        return definition.releaseSeconds;

      case EnvelopeKind.OneShot:
        return 0;
    }
  }

  private getReleaseCurve(definition: EnvelopeDefinition): IEnvelopeCurveDefinition {
    switch (definition.kind) {
      case EnvelopeKind.Adsr:
      case EnvelopeKind.Ahdsr:
      case EnvelopeKind.Ar:
      case EnvelopeKind.Asr:
        return definition.releaseCurve;

      case EnvelopeKind.OneShot:
        return {
          kind: EnvelopeCurveKind.Instant,
        };
    }
  }

  private assertSeconds(value: EnvelopeSecondsValue): void {
    if (!this.secondsSpecification.isSatisfiedBy(value)) {
      throw new Error(`timeSeconds must be finite and greater than or equal to 0. Actual value: ${String(value)}.`);
    }
  }

  private assertVelocity(value: EnvelopeLevelValue): void {
    if (!this.velocitySpecification.isSatisfiedBy(value)) {
      throw new Error(`velocity must be between 0 and 1. Actual value: ${String(value)}.`);
    }
  }
}

  public static clamp01(value: number): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value < 0) {
      return 0;
    }

    if (value > 1) {
      return 1;
    }

    return value;
  }

  public static safeTimeConstant(durationSeconds: number): number {
    if (!Number.isFinite(durationSeconds) || durationSeconds <= 0) {
      return 0.001;
    }

    return Math.max(0.001, durationSeconds / 5);
  }

  public static levelAtRelease(parameterValue: number): number {
    return WebAudioEnvelopeMath.clamp01(parameterValue);
  }
}

import {
  OscillatorWaveformKind,
  type AudioSampleValue,
  type IBasicOscillatorDefinition,
} from "@monorepo/music-core-synthesizer-oscillators-contracts";

import { OscillatorMath } from "../math/index.js";
import type { IOscillatorAlgorithm } from "./IOscillatorAlgorithm.js";

export class BasicOscillatorAlgorithm
  implements IOscillatorAlgorithm<IBasicOscillatorDefinition> {
  public sample(
    definition: IBasicOscillatorDefinition,
    phase01: number,
  ): AudioSampleValue {
    const phase = OscillatorMath.wrap01(phase01);

    switch (definition.waveform) {
      case OscillatorWaveformKind.Sine:
        return Math.sin(phase * OscillatorMath.twoPi);

      case OscillatorWaveformKind.Square:
        return phase < 0.5 ? 1 : -1;

      case OscillatorWaveformKind.Sawtooth:
        return phase * 2 - 1;

      case OscillatorWaveformKind.Triangle:
        return 1 - 4 * Math.abs(phase - 0.5);
    }
  }
}

// FILE: src\math\OscillatorMath.ts

export class OscillatorMath {
  public static readonly twoPi = Math.PI * 2;

  public static clampBipolar(value: number): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value < -1) {
      return -1;
    }

    if (value > 1) {
      return 1;
    }

    return value;
  }

  public static clampUnipolar(value: number): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value < 0) {
      return 0;
    }

    if (value > 1) {
      return 1;
    }

    return value;
  }

  public static wrap01(value: number): number {
    if (!Number.isFinite(value)) {
      return 0;
    }

    const wrapped = value % 1;

    return wrapped < 0 ? wrapped + 1 : wrapped;
  }

  public static radiansToCycles(radians: number): number {
    return OscillatorMath.wrap01(radians / OscillatorMath.twoPi);
  }

  public static centsToFrequencyRatio(cents: number): number {
    return Math.pow(2, cents / 1_200);
  }

  public static frequencyWithDetune(
    frequencyHz: number,
    detuneCents: number,
  ): number {
    return frequencyHz * OscillatorMath.centsToFrequencyRatio(detuneCents);
  }

  public static phaseIncrement(
    frequencyHz: number,
    sampleRateHz: number,
  ): number {
    return frequencyHz / sampleRateHz;
  }
}

  private activePhaseRadians: PhaseRadiansValue;
  private active = false;

  public constructor(
    public readonly definition: OscillatorDefinition,
    private readonly algorithm: IOscillatorAlgorithm,
    private readonly definitionValidator: OscillatorDefinitionValidator,
  ) {
    this.definitionValidator.validate(definition);

    this.activeFrequencyHz = this.getDefinitionFrequency(definition);
    this.activeDetuneCents = this.getDefinitionDetune(definition);
    this.activePhaseRadians = this.getDefinitionPhase(definition);
    this.phase01 = OscillatorMath.radiansToCycles(this.activePhaseRadians);
  }

  public noteOn(request: IOscillatorNoteOnRequest): void {
    this.assertTime(request.timeSeconds);
    this.assertFrequency(request.frequencyHz);
    this.assertVelocity(request.velocity);

    this.activeFrequencyHz = request.frequencyHz;
    this.activeDetuneCents = this.getDefinitionDetune(this.definition);
    this.activePhaseRadians = this.getDefinitionPhase(this.definition);
    this.phase01 = OscillatorMath.radiansToCycles(this.activePhaseRadians);
    this.active = true;
  }

  public noteOff(request: IOscillatorNoteOffRequest): void {
    this.assertTime(request.timeSeconds);

    this.active = false;
  }

  public sample(request: IOscillatorSampleRequest): IOscillatorSample {
    this.assertTime(request.timeSeconds);
    this.assertSampleRate(request.sampleRateHz);

    if (!this.active) {
      return {
        timeSeconds: request.timeSeconds,
        value: 0,
      };
    }

    const value = this.nextSample(request.sampleRateHz);

    return {
      timeSeconds: request.timeSeconds,
      value,
    };
  }

  public snapshot(): IOscillatorStateSnapshot {
    return {
      definition: this.definition,
      frequencyHz: this.activeFrequencyHz,
      detuneCents: this.activeDetuneCents,
      phaseRadians: this.activePhaseRadians,
      gain: this.definition.gain,
      isActive: this.active,
    };
  }

  public reset(): void {
    this.active = false;
    this.activeFrequencyHz = this.getDefinitionFrequency(this.definition);
    this.activeDetuneCents = this.getDefinitionDetune(this.definition);
    this.activePhaseRadians = this.getDefinitionPhase(this.definition);
    this.phase01 = OscillatorMath.radiansToCycles(this.activePhaseRadians);
  }

  private nextSample(sampleRateHz: number): AudioSampleValue {
    const raw = this.algorithm.sample(this.definition, this.phase01);
    const value = OscillatorMath.clampBipolar(raw * this.definition.gain);
    const effectiveFrequency = OscillatorMath.frequencyWithDetune(
      this.activeFrequencyHz,
      this.activeDetuneCents,
    );

    this.phase01 = OscillatorMath.wrap01(
      this.phase01 + OscillatorMath.phaseIncrement(effectiveFrequency, sampleRateHz),
    );

    return value;
  }

  private getDefinitionFrequency(definition: OscillatorDefinition): FrequencyHzValue {
    switch (definition.kind) {
      case OscillatorKind.Basic:
      case OscillatorKind.Pulse:
      case OscillatorKind.Wavetable:
        return definition.frequencyHz;

      case OscillatorKind.Sample:
        return definition.playbackFrequencyHz;
    }
  }

  private getDefinitionDetune(definition: OscillatorDefinition): DetuneCentsValue {
    switch (definition.kind) {
      case OscillatorKind.Basic:
      case OscillatorKind.Pulse:
      case OscillatorKind.Wavetable:
        return definition.detuneCents;

      case OscillatorKind.Sample:
        return 0;
    }
  }

  private getDefinitionPhase(definition: OscillatorDefinition): PhaseRadiansValue {
    switch (definition.kind) {
      case OscillatorKind.Basic:
      case OscillatorKind.Pulse:
      case OscillatorKind.Wavetable:
        return definition.phaseRadians;

      case OscillatorKind.Sample:
        return 0;
    }
  }

  private assertTime(value: number): void {
    if (!Number.isFinite(value) || value < 0) {
      throw new Error(`timeSeconds must be finite and greater than or equal to 0. Actual value: ${String(value)}.`);
    }
  }

  private assertFrequency(value: number): void {
    if (!Number.isFinite(value) || value <= 0) {
      throw new Error(`frequencyHz must be finite and greater than 0. Actual value: ${String(value)}.`);
    }
  }

  private assertVelocity(value: number): void {
    if (!Number.isFinite(value) || value < 0 || value > 1) {
      throw new Error(`velocity must be between 0 and 1. Actual value: ${String(value)}.`);
    }
  }

  private assertSampleRate(value: number): void {
    if (!Number.isFinite(value) || value <= 0) {
      throw new Error(`sampleRateHz must be finite and greater than 0. Actual value: ${String(value)}.`);
    }
  }
}

  private active = false;

  public readonly outputNode: IWebAudioConnectable;

  private readonly waveformMapper = new OscillatorWaveformMapper();

  public constructor(
    private readonly context: IWebAudioContextLike,
    public readonly definition: OscillatorDefinition,
    private readonly definitionValidator: WebAudioOscillatorDefinitionValidator,
    private readonly wavetableProvider: IWebAudioWavetableProvider,
    private readonly sampleBufferProvider: IWebAudioSampleBufferProvider,
  ) {
    this.definitionValidator.validate(definition);

    this.outputNode = this.context.createGain();

    this.activeFrequencyHz = this.getDefinitionFrequency(definition);
    this.activeDetuneCents = this.getDefinitionDetune(definition);
    this.activePhaseRadians = this.getDefinitionPhase(definition);

    this.setOutputGain(definition.gain);
  }

  public noteOn(request: IOscillatorNoteOnRequest): void {
    this.assertTime(request.timeSeconds);
    this.assertFrequency(request.frequencyHz);
    this.assertVelocity(request.velocity);

    this.stopActiveSource(request.timeSeconds);

    const source = this.createSource(this.definition, request);

    source.connect(this.outputNode);
    source.start(request.timeSeconds);

    this.activeSource = source;
    this.activeFrequencyHz = request.frequencyHz;
    this.activeDetuneCents = this.getDefinitionDetune(this.definition);
    this.activePhaseRadians = this.getDefinitionPhase(this.definition);
    this.active = true;

    this.setOutputGain(this.definition.gain * request.velocity);
  }

  public noteOff(request: IOscillatorNoteOffRequest): void {
    this.assertTime(request.timeSeconds);

    this.stopActiveSource(request.timeSeconds);
    this.active = false;
  }

  public sample(request: IOscillatorSampleRequest): IOscillatorSample {
    this.assertTime(request.timeSeconds);
    this.assertSampleRate(request.sampleRateHz);

    return {
      timeSeconds: request.timeSeconds,
      value: 0,
    };
  }

  public snapshot(): IOscillatorStateSnapshot {
    return {
      definition: this.definition,
      frequencyHz: this.activeFrequencyHz,
      detuneCents: this.activeDetuneCents,
      phaseRadians: this.activePhaseRadians,
      gain: this.definition.gain,
      isActive: this.active,
    };
  }

  public reset(): void {
    this.stopActiveSource(0);
    this.active = false;
    this.activeFrequencyHz = this.getDefinitionFrequency(this.definition);
    this.activeDetuneCents = this.getDefinitionDetune(this.definition);
    this.activePhaseRadians = this.getDefinitionPhase(this.definition);

    this.setOutputGain(this.definition.gain);
  }

  public connect(destination: IWebAudioConnectable): void {
    this.outputNode.connect(destination);
  }

  public disconnect(): void {
    this.outputNode.disconnect();
  }

  private createSource(
    definition: OscillatorDefinition,
    request: IOscillatorNoteOnRequest,
  ): IWebAudioSourceNodeLike {
    switch (definition.kind) {
      case OscillatorKind.Basic:
        return this.createBasicSource(definition, request);

      case OscillatorKind.Pulse:
        return this.createPulseSource(definition, request);

      case OscillatorKind.Wavetable:
        return this.createWavetableSource(definition, request);

      case OscillatorKind.Sample:
        return this.createSampleSource(definition, request);
    }
  }

  private createBasicSource(
    definition: IBasicOscillatorDefinition,
    request: IOscillatorNoteOnRequest,
  ): IWebAudioOscillatorNodeLike {
    const source = this.context.createOscillator();

    source.type = this.waveformMapper.map(definition.waveform);
    source.frequency.setValueAtTime(request.frequencyHz, request.timeSeconds);
    source.detune.setValueAtTime(definition.detuneCents, request.timeSeconds);

    return source;
  }

  private createPulseSource(
    definition: IPulseOscillatorDefinition,
    request: IOscillatorNoteOnRequest,
  ): IWebAudioOscillatorNodeLike {
    const source = this.context.createOscillator();

    source.type = "square";
    source.frequency.setValueAtTime(request.frequencyHz, request.timeSeconds);
    source.detune.setValueAtTime(definition.detuneCents, request.timeSeconds);

    return source;
  }

  private createWavetableSource(
    definition: IWavetableOscillatorDefinition,
    request: IOscillatorNoteOnRequest,
  ): IWebAudioOscillatorNodeLike {
    const source = this.context.createOscillator();
    const periodicWave = this.wavetableProvider.get(definition.tableId);

    source.setPeriodicWave(periodicWave);
    source.frequency.setValueAtTime(request.frequencyHz, request.timeSeconds);
    source.detune.setValueAtTime(definition.detuneCents, request.timeSeconds);

    return source;
  }

  private createSampleSource(
    definition: ISampleOscillatorDefinition,
    request: IOscillatorNoteOnRequest,
  ): IWebAudioBufferSourceNodeLike {
    const source = this.context.createBufferSource();
    const buffer = this.sampleBufferProvider.get(definition.sampleId);

    source.buffer = buffer;
    source.loop = definition.loop;
    source.playbackRate.setValueAtTime(
      request.frequencyHz / definition.rootFrequencyHz,
      request.timeSeconds,
    );

    return source;
  }

  private stopActiveSource(timeSeconds: number): void {
    if (this.activeSource === null) {
      return;
    }

    this.activeSource.stop(timeSeconds);
    this.activeSource.disconnect();
    this.activeSource = null;
  }

  private setOutputGain(value: number): void {
    const gainNode = this.outputNode as {
      readonly gain?: {
        value: number;
      };
    };

    if (gainNode.gain !== undefined) {
      gainNode.gain.value = value;
    }
  }

  private getDefinitionFrequency(definition: OscillatorDefinition): FrequencyHzValue {
    switch (definition.kind) {
      case OscillatorKind.Basic:
      case OscillatorKind.Pulse:
      case OscillatorKind.Wavetable:
        return definition.frequencyHz;

      case OscillatorKind.Sample:
        return definition.playbackFrequencyHz;
    }
  }

  private getDefinitionDetune(definition: OscillatorDefinition): DetuneCentsValue {
    switch (definition.kind) {
      case OscillatorKind.Basic:
      case OscillatorKind.Pulse:
      case OscillatorKind.Wavetable:
        return definition.detuneCents;

      case OscillatorKind.Sample:
        return 0;
    }
  }

  private getDefinitionPhase(definition: OscillatorDefinition): PhaseRadiansValue {
    switch (definition.kind) {
      case OscillatorKind.Basic:
      case OscillatorKind.Pulse:
      case OscillatorKind.Wavetable:
        return definition.phaseRadians;

      case OscillatorKind.Sample:
        return 0;
    }
  }

  private assertTime(value: number): void {
    if (!Number.isFinite(value) || value < 0) {
      throw new Error(
        `timeSeconds must be finite and greater than or equal to 0. Actual value: ${String(value)}.`,
      );
    }
  }

  private assertFrequency(value: number): void {
    if (!Number.isFinite(value) || value <= 0) {
      throw new Error(
        `frequencyHz must be finite and greater than 0. Actual value: ${String(value)}.`,
      );
    }
  }

  private assertVelocity(value: number): void {
    if (!Number.isFinite(value) || value < 0 || value > 1) {
      throw new Error(
        `velocity must be between 0 and 1. Actual value: ${String(value)}.`,
      );
    }
  }

  private assertSampleRate(value: number): void {
    if (!Number.isFinite(value) || value <= 0) {
      throw new Error(
        `sampleRateHz must be finite and greater than 0. Actual value: ${String(value)}.`,
      );
    }
  }
}

import {

  type OscillatorDefinition,
  type IOscillatorDefinitionValidator,
} from "@monorepo/music-core-synthesizer-oscillators-contracts";

export class WebAudioOscillatorDefinitionValidator
  implements IOscillatorDefinitionValidator {
  public validate(definition: OscillatorDefinition): void {
    this.assertGain(definition.gain);

    switch (definition.kind) {
      case OscillatorKind.Basic:
        this.assertWaveform(definition.waveform);
        this.assertFrequency(definition.frequencyHz, "frequencyHz");
        this.assertFinite(definition.detuneCents, "detuneCents");
        this.assertFinite(definition.phaseRadians, "phaseRadians");
        return;

      case OscillatorKind.Pulse:
        this.assertFrequency(definition.frequencyHz, "frequencyHz");
        this.assertFinite(definition.detuneCents, "detuneCents");
        this.assertFinite(definition.phaseRadians, "phaseRadians");
        this.assertUnipolar(definition.pulseWidth, "pulseWidth");
        return;

      case OscillatorKind.Wavetable:
        this.assertNonEmptyString(definition.tableId, "tableId");
        this.assertFrequency(definition.frequencyHz, "frequencyHz");
        this.assertFinite(definition.detuneCents, "detuneCents");
        this.assertFinite(definition.phaseRadians, "phaseRadians");
        return;

      case OscillatorKind.Sample:
        this.assertNonEmptyString(definition.sampleId, "sampleId");
        this.assertFrequency(definition.rootFrequencyHz, "rootFrequencyHz");
        this.assertFrequency(definition.playbackFrequencyHz, "playbackFrequencyHz");
        return;
    }
  }

  private assertGain(value: number): void {
    if (!Number.isFinite(value) || value < 0) {
      throw new Error(`gain must be finite and greater than or equal to 0. Actual value: ${String(value)}.`);
    }
  }

  private assertFrequency(
    value: number,
    name: string,
  ): void {
    if (!Number.isFinite(value) || value <= 0) {
      throw new Error(`${name} must be finite and greater than 0. Actual value: ${String(value)}.`);
    }
  }

  private assertFinite(
    value: number,
    name: string,
  ): void {
    if (!Number.isFinite(value)) {
      throw new Error(`${name} must be finite. Actual value: ${String(value)}.`);
    }
  }

  private assertUnipolar(
    value: number,
    name: string,
  ): void {
    if (!Number.isFinite(value) || value < 0 || value > 1) {
      throw new Error(`${name} must be between 0 and 1. Actual value: ${String(value)}.`);
    }
  }

  private assertNonEmptyString(
    value: string,
    name: string,
  ): void {
    if (value.trim().length === 0) {
      throw new Error(`${name} must not be empty.`);
    }
  }

  private assertWaveform(value: string): void {
    if (!Object.values(OscillatorWaveformKind).includes(value as never)) {
      throw new Error(`Unsupported oscillator waveform: ${value}.`);
    }
  }

            min="-1200"

            step="1"

            @input=${this.onDetuneChanged}
          />
        </div>

        ${label}
      </option>
    `;

    const select = event.target as HTMLSelectElement;

    this.patchBasicOscillator({
      waveform: select.value as IBasicOscillatorDefinition["waveform"],
    });
  };

  private readonly onGainChanged = (event: Event): void => {
    const input = event.target as HTMLInputElement;

    this.patchBasicOscillator({
      gain: Number(input.value),
    });
  };

  private readonly onDetuneChanged = (event: Event): void => {
    const input = event.target as HTMLInputElement;

    this.patchBasicOscillator({
      detuneCents: Number(input.value),
    });
  };

  private readonly onMidiNoteOn = (
    event: CustomEvent<IOscillatorMidiNoteOnDetail>,
  ): void => {
    this.dispatchEvent(
      new CustomEvent<IOscillatorMidiNoteOnDetail>(
        OscillatorControlEvent.MidiNoteOnReceived,
        {
          bubbles: true,
          composed: true,
          detail: event.detail,
        },
      ),
    );
  };

  private readonly onMidiNoteOff = (
    event: CustomEvent<IOscillatorMidiNoteOffDetail>,
  ): void => {
    this.dispatchEvent(
      new CustomEvent<IOscillatorMidiNoteOffDetail>(
        OscillatorControlEvent.MidiNoteOffReceived,
        {
          bubbles: true,
          composed: true,
          detail: event.detail,
        },
      ),
    );
  };

  private patchBasicOscillator(patch: Partial<IBasicOscillatorDefinition>): void {
    if (this.definition === null || this.definition.kind !== OscillatorKind.Basic) {
      return;
    }

    this.definition = {
      ...this.definition,
      ...patch,
      kind: OscillatorKind.Basic,
    };

    this.dispatchEvent(
      new CustomEvent<IOscillatorDefinitionChangedDetail>(
        OscillatorControlEvent.DefinitionChanged,
        {
          bubbles: true,
          composed: true,
          detail: {
            definition: this.definition,
          },
        },
      ),
    );
  }
}

export interface IParameterScale<TValue> {
  fromNormalized(value: UnipolarNormalizedValue): TValue;
  toNormalized(value: TValue): UnipolarNormalizedValue;
}

export class LinearParameterScale implements IParameterScale<number> {
  public constructor(
    private readonly minimum: number,
    private readonly maximum: number,
  ) {
    if (!Number.isFinite(minimum)) {
      throw new Error("minimum must be finite.");
    }

    if (!Number.isFinite(maximum)) {
      throw new Error("maximum must be finite.");
    }

    if (maximum <= minimum) {
      throw new Error("maximum must be greater than minimum.");
    }
  }

  public fromNormalized(value: UnipolarNormalizedValue): number {
    return this.minimum + value * (this.maximum - this.minimum);
  }

  public toNormalized(value: number): UnipolarNormalizedValue {
    return (value - this.minimum) / (this.maximum - this.minimum);
  }
}

export class BipolarLinearParameterScale implements IParameterScale<number> {
  public constructor(
    private readonly negativeMaximum: number,
    private readonly positiveMaximum: number,
  ) {
    if (!Number.isFinite(negativeMaximum)) {
      throw new Error("negativeMaximum must be finite.");
    }

    if (!Number.isFinite(positiveMaximum)) {
      throw new Error("positiveMaximum must be finite.");
    }

    if (negativeMaximum >= 0) {
      throw new Error("negativeMaximum must be less than 0.");
    }

    if (positiveMaximum <= 0) {
      throw new Error("positiveMaximum must be greater than 0.");
    }
  }

  public fromNormalized(value: UnipolarNormalizedValue): number {
    return this.negativeMaximum + value * (this.positiveMaximum - this.negativeMaximum);
  }

  public toNormalized(value: number): UnipolarNormalizedValue {
    return (value - this.negativeMaximum) / (this.positiveMaximum - this.negativeMaximum);
  }
}

export class LogarithmicFrequencyScale implements IParameterScale<FrequencyHzValue> {
  public constructor(
    private readonly minimumHz: FrequencyHzValue,
    private readonly maximumHz: FrequencyHzValue,
  ) {
    FrequencyHz.create(minimumHz);
    FrequencyHz.create(maximumHz);

    if (maximumHz <= minimumHz) {
      throw new Error("maximumHz must be greater than minimumHz.");
    }
  }

  public fromNormalized(value: UnipolarNormalizedValue): FrequencyHzValue {
    return this.minimumHz * Math.pow(this.maximumHz / this.minimumHz, value);
  }

  public toNormalized(value: FrequencyHzValue): UnipolarNormalizedValue {
    return Math.log(value / this.minimumHz) / Math.log(this.maximumHz / this.minimumHz);
  }
}

export class ExponentialParameterScale implements IParameterScale<number> {
  public constructor(
    private readonly minimum: number,
    private readonly maximum: number,
    private readonly exponent: number,
  ) {
    if (!Number.isFinite(minimum)) {
      throw new Error("minimum must be finite.");
    }

    if (!Number.isFinite(maximum)) {
      throw new Error("maximum must be finite.");
    }

    if (maximum <= minimum) {
      throw new Error("maximum must be greater than minimum.");
    }

    if (!Number.isFinite(exponent) || exponent <= 0) {
      throw new Error("exponent must be finite and greater than 0.");
    }
  }

  public fromNormalized(value: UnipolarNormalizedValue): number {
    return this.minimum + Math.pow(value, this.exponent) * (this.maximum - this.minimum);
  }

  public toNormalized(value: number): UnipolarNormalizedValue {
    return Math.pow((value - this.minimum) / (this.maximum - this.minimum), 1 / this.exponent);
  }
}

export class SteppedParameterScale implements IParameterScale<number> {
  public constructor(
    private readonly minimum: number,
    private readonly maximum: number,
    private readonly stepCount: number,
  ) {
    if (!Number.isFinite(minimum)) {
      throw new Error("minimum must be finite.");
    }

    if (!Number.isFinite(maximum)) {
      throw new Error("maximum must be finite.");
    }

    if (maximum <= minimum) {
      throw new Error("maximum must be greater than minimum.");
    }

    if (!Number.isInteger(stepCount) || stepCount <= 1) {
      throw new Error("stepCount must be an integer greater than 1.");
    }
  }

  public fromNormalized(value: UnipolarNormalizedValue): number {
    const index = Math.round(value * (this.stepCount - 1));
    const normalizedStep = index / (this.stepCount - 1);

    return this.minimum + normalizedStep * (this.maximum - this.minimum);
  }

  public toNormalized(value: number): UnipolarNormalizedValue {
    const raw = (value - this.minimum) / (this.maximum - this.minimum);
    const index = Math.round(raw * (this.stepCount - 1));

    return index / (this.stepCount - 1);
  }
}

  }

  public isSatisfiedBy(candidate: number): boolean {

  }

  public isSatisfiedBy(candidate: number): boolean {

  }

export class MinimumNumberSpecification extends Specification<number> {
  public constructor(private readonly minimum: number) {
    super();

    if (!Number.isFinite(minimum)) {
      throw new Error("minimum must be finite.");
    }
  }

  public isSatisfiedBy(candidate: number): boolean {
    return Number.isFinite(candidate) && candidate >= this.minimum;
  }
}

export class MaximumNumberSpecification extends Specification<number> {
  public constructor(private readonly maximum: number) {
    super();

    if (!Number.isFinite(maximum)) {
      throw new Error("maximum must be finite.");
    }
  }

  public isSatisfiedBy(candidate: number): boolean {
    return Number.isFinite(candidate) && candidate <= this.maximum;
  }
}

export class RangeNumberSpecification extends Specification<number> {
  private readonly specification: ISpecification<number>;

  public constructor(
    minimum: number,
    maximum: number,
  ) {
    super();

    if (!Number.isFinite(minimum)) {
      throw new Error("minimum must be finite.");
    }

    if (!Number.isFinite(maximum)) {
      throw new Error("maximum must be finite.");
    }

    if (maximum < minimum) {
      throw new Error("maximum must be greater than or equal to minimum.");
    }

    this.specification = new MinimumNumberSpecification(minimum).and(
      new MaximumNumberSpecification(maximum),
    );
  }

  public isSatisfiedBy(candidate: number): boolean {
    return this.specification.isSatisfiedBy(candidate);
  }
}

  }

    return new FiniteNumberSpecification().and(new NonNegativeNumberSpecification());
  }

  public static integerFinite(): ISpecification<number> {
    return new FiniteNumberSpecification().and(new IntegerNumberSpecification());
  }

  public static range(
    minimum: number,
    maximum: number,
  ): ISpecification<number> {
    return new FiniteNumberSpecification().and(
      new RangeNumberSpecification(minimum, maximum),
    );
  }

  public static integerRange(
    minimum: number,
    maximum: number,
  ): ISpecification<number> {
    return new FiniteNumberSpecification()
      .and(new IntegerNumberSpecification())
      .and(new RangeNumberSpecification(minimum, maximum));
  }
}

export class WebAudioEnvelopeScheduler implements IWebAudioEnvelopeScheduler {
  public noteOn(
    param: AudioParam,
    env: IEnvelopeDefinition,
    velocity: Normalized,
    at: Seconds,
  ): void {
    WebAudioGuards.normalized(velocity, "velocity");

    const adsr = WebAudioGuards.adsrEnvelope(env);
    const attackEnd = at + adsr.attackSeconds;
    const decayEnd = attackEnd + adsr.decaySeconds;
    const sustain = adsr.sustainLevel * velocity;

    param.cancelScheduledValues(at);
    param.setValueAtTime(0, at);
    param.linearRampToValueAtTime(velocity, attackEnd);
    param.linearRampToValueAtTime(sustain, decayEnd);
  }

  public noteOff(
    param: AudioParam,
    env: IEnvelopeDefinition,
    at: Seconds,
  ): void {
    const adsr = WebAudioGuards.adsrEnvelope(env);
    const releaseEnd = at + adsr.releaseSeconds;

    param.cancelScheduledValues(at);
    param.setValueAtTime(param.value, at);
    param.linearRampToValueAtTime(0, releaseEnd);
  }

  public cancel(param: AudioParam, at: Seconds): void {
    param.cancelScheduledValues(at);
  }
}

export class WebAudioWhiteNoiseSourceFactory implements IWebAudioNoiseSourceFactory {
  public constructor(private readonly durationSeconds = 2) {
    if (!Number.isFinite(durationSeconds) || durationSeconds <= 0) {
      throw new Error("Noise buffer duration must be greater than zero.");
    }
  }

  public create(context: AudioContext): AudioNode {
    const frameCount = Math.ceil(context.sampleRate * this.durationSeconds);
    const buffer = context.createBuffer(1, frameCount, context.sampleRate);
    const data = buffer.getChannelData(0);

    for (let i = 0; i < data.length; i++) {
      data[i] = Math.random() * 2 - 1;
    }

    const source = context.createBufferSource();
    source.buffer = buffer;
    source.loop = true;
    source.start();

    return source;
  }
}

export class WebAudioVoiceGraphBuilder implements IWebAudioVoiceGraphBuilder {
  public constructor(
    private readonly noiseSourceFactory: IWebAudioNoiseSourceFactory,
  ) {}

  public build(request: {
    readonly context: AudioContext;
    readonly destination: AudioNode;
    readonly definition: IVoiceDefinition;
    readonly frequencyHz: Hertz;
  }): IWebAudioVoiceGraph {
    WebAudioGuards.frequency(request.frequencyHz, "frequencyHz");
    WebAudioGuards.voiceDefinition(request.definition);

    const filter = this.createFilterOrNull(
      request.context,
      request.definition.filter,
    );

    const distortion = this.createDistortionOrNull(
      request.context,
      request.definition.distortion,
    );

    const vca = request.context.createGain();
    vca.gain.setValueAtTime(0, request.context.currentTime);

    const firstProcessor = filter ?? distortion ?? vca;

    const oscillatorMixer = request.context.createGain();
    oscillatorMixer.gain.setValueAtTime(1, request.context.currentTime);

    const oscillators = request.definition.oscillators.map((definition) => {
      const oscillator = request.context.createOscillator();
      oscillator.type = definition.waveform;
      oscillator.frequency.setValueAtTime(
        request.frequencyHz,
        request.context.currentTime,
      );
      oscillator.detune.setValueAtTime(
        definition.detuneCents,
        request.context.currentTime,
      );

      const gain = request.context.createGain();
      gain.gain.setValueAtTime(definition.gain, request.context.currentTime);

      oscillator.connect(gain);
      gain.connect(oscillatorMixer);

      return oscillator;
    });

    oscillatorMixer.connect(firstProcessor);

    if (filter !== null && distortion !== null) {
      filter.connect(distortion);
      distortion.connect(vca);
    } else if (filter !== null) {
      filter.connect(vca);
    } else if (distortion !== null) {
      distortion.connect(vca);
    }

    const noise = this.createNoiseOrNull(
      request.context,
      request.definition,
      firstProcessor,
    );

    vca.connect(request.destination);

    return {
      output: vca,
      oscillators,
      vca,
      filter,
      distortion,
      noise,
    };
  }

  private createFilterOrNull(
    context: AudioContext,
    definition: IFilterDefinition,
  ): BiquadFilterNode | null {
    if (!definition.enabled) {
      return null;
    }

    WebAudioGuards.frequency(definition.cutoffHz, "filter cutoffHz");

    if (!Number.isFinite(definition.resonanceQ) || definition.resonanceQ < 0) {
      throw new Error("filter resonanceQ must be >= 0.");
    }

    const node = context.createBiquadFilter();

    switch (definition.type) {
      case FilterType.LowPass:
        node.type = "lowpass";
        break;
      case FilterType.HighPass:
        node.type = "highpass";
        break;
      case FilterType.BandPass:
        node.type = "bandpass";
        break;
      default:
        throw new Error("Unsupported filter type.");
    }

    node.frequency.setValueAtTime(definition.cutoffHz, context.currentTime);
    node.Q.setValueAtTime(definition.resonanceQ, context.currentTime);

    return node;
  }

  private createDistortionOrNull(
    context: AudioContext,
    definition: IDistortionDefinition,
  ): WaveShaperNode | null {
    if (!definition.enabled || definition.type === DistortionType.None) {
      return null;
    }

    WebAudioGuards.normalized(definition.amount, "distortion amount");

    const node = context.createWaveShaper();
    node.curve = WebAudioDistortionCurveFactory.create(definition);
    node.oversample = "4x";

    return node;
  }

  private createNoiseOrNull(
    context: AudioContext,
    definition: IVoiceDefinition,
    destination: AudioNode,
  ): AudioNode | null {
    if (!definition.noise.enabled || definition.noise.gain <= 0) {
      return null;
    }

    WebAudioGuards.normalized(definition.noise.gain, "noise gain");

    const source = this.noiseSourceFactory.create(context);
    const gain = context.createGain();

    gain.gain.setValueAtTime(definition.noise.gain, context.currentTime);

    source.connect(gain);
    gain.connect(destination);

    return source;
  }
}

export class WebAudioVoiceHandle implements IVoiceHandle {
  public readonly id: string;
  public readonly note: MidiNoteNumber;

  private released = false;
  private disposed = false;
  private started = false;

  public constructor(
    private readonly graph: IWebAudioVoiceGraph,
    private readonly envelopeScheduler: IWebAudioEnvelopeScheduler,
    private readonly envelope: IEnvelopeDefinition,
    note: MidiNoteNumber,
    private readonly velocity: Normalized,
    startedAt: Seconds,
  ) {
    this.id = `${note}:${startedAt}`;
    this.note = note;
  }

  public get isReleased(): boolean {
    return this.released;
  }

  public start(at: Seconds): void {
    this.throwIfDisposed();

    if (this.started) {
      throw new Error("Voice has already been started.");
    }

    this.started = true;

    this.envelopeScheduler.noteOn(
      this.graph.vca.gain,
      this.envelope,
      this.velocity,
      at,
    );

    for (const oscillator of this.graph.oscillators) {
      oscillator.start(at);
    }
  }

  public release(at: Seconds): void {
    if (this.disposed || this.released) {
      return;
    }

    this.released = true;

    this.envelopeScheduler.noteOff(this.graph.vca.gain, this.envelope, at);

    const adsr = WebAudioGuards.adsrEnvelope(this.envelope);
    const stopAt = at + adsr.releaseSeconds + 0.02;

    for (const oscillator of this.graph.oscillators) {
      try {
        oscillator.stop(stopAt);
      } catch {
        // already stopped
      }
    }

    const first = this.graph.oscillators[0];

    if (first) {
      first.addEventListener("ended", () => this.dispose(), { once: true });
      return;
    }

    this.dispose();
  }

  public dispose(): void {
    if (this.disposed) {
      return;
    }

    this.disposed = true;

    for (const oscillator of this.graph.oscillators) {
      oscillator.disconnect();
    }

    this.graph.vca.disconnect();
    this.graph.filter?.disconnect();
    this.graph.distortion?.disconnect();
    this.graph.noise?.disconnect();
  }

  private throwIfDisposed(): void {
    if (this.disposed) {
      throw new Error("Voice has already been disposed.");
    }
  }
}

export class WebAudioDistortionCurveFactory {
  private static readonly sampleCount = 2048;

  public static create(
    definition: IDistortionDefinition,
  ): Float32Array<ArrayBuffer> {
    WebAudioGuards.normalized(definition.amount, "distortion amount");

    switch (definition.type) {
      case DistortionType.SoftClip:
        return this.softClip(definition.amount);
      case DistortionType.HardClip:
        return this.hardClip(definition.amount);
      case DistortionType.Tanh:
        return this.tanh(definition.amount);
      case DistortionType.None:
        return this.identity();
      default:
        throw new Error("Unsupported distortion type.");
    }
  }

  private static createCurve(): Float32Array<ArrayBuffer> {
    return new Float32Array(
      new ArrayBuffer(this.sampleCount * Float32Array.BYTES_PER_ELEMENT),
    );
  }

  private static identity(): Float32Array<ArrayBuffer> {
    const curve = this.createCurve();

    for (let i = 0; i < curve.length; i++) {
      const x = (i * 2) / (curve.length - 1) - 1;
      curve[i] = x;
    }

    return curve;
  }

  private static softClip(amount: Normalized): Float32Array<ArrayBuffer> {
    const curve = this.createCurve();
    const drive = 1 + amount * 20;

    for (let i = 0; i < curve.length; i++) {
      const x = (i * 2) / (curve.length - 1) - 1;
      curve[i] = x / (1 + Math.abs(x * drive));
    }

    return curve;
  }

  private static hardClip(amount: Normalized): Float32Array<ArrayBuffer> {
    const curve = this.createCurve();
    const threshold = Math.max(0.05, 1 - amount);

    for (let i = 0; i < curve.length; i++) {
      const x = (i * 2) / (curve.length - 1) - 1;
      curve[i] = Math.max(-threshold, Math.min(threshold, x)) / threshold;
    }

    return curve;
  }

  private static tanh(amount: Normalized): Float32Array<ArrayBuffer> {
    const curve = this.createCurve();
    const drive = 1 + amount * 40;

    for (let i = 0; i < curve.length; i++) {
      const x = (i * 2) / (curve.length - 1) - 1;
      curve[i] = Math.tanh(x * drive);
    }

    return curve;
  }
}

class WebAudioGuards {
  public static normalized(value: number, name: string): void {
    if (!Number.isFinite(value) || value < 0 || value > 1) {
      throw new Error(`${name} must be in range 0..1.`);
    }
  }

  public static frequency(value: number, name: string): void {
    if (!Number.isFinite(value) || value <= 0) {
      throw new Error(`${name} must be greater than zero.`);
    }
  }

  public static adsrEnvelope(env: IEnvelopeDefinition): IAdsrEnvelopeDefinition {
    if (env.type !== ADSR_ENVELOPE) {
      throw new Error(`Unsupported envelope type '${env.type}'.`);
    }

    const adsr = env as IAdsrEnvelopeDefinition;

    this.nonNegative(adsr.attackSeconds, "Envelope attackSeconds");
    this.nonNegative(adsr.decaySeconds, "Envelope decaySeconds");
    this.nonNegative(adsr.releaseSeconds, "Envelope releaseSeconds");
    this.normalized(adsr.sustainLevel, "Envelope sustainLevel");

    return adsr;
  }

  public static voiceDefinition(definition: IVoiceDefinition): void {
    if (definition.oscillators.length === 0) {
      throw new Error("Voice definition must contain at least one oscillator.");
    }

    for (const oscillator of definition.oscillators) {
      this.normalized(oscillator.gain, "oscillator gain");
    }

    this.adsrEnvelope(definition.envelope);
    this.normalized(definition.noise.gain, "noise gain");
    this.normalized(definition.distortion.amount, "distortion amount");
  }

  private static nonNegative(value: number, name: string): void {
    if (!Number.isFinite(value) || value < 0) {
      throw new Error(`${name} must be >= 0.`);
    }
  }
}
