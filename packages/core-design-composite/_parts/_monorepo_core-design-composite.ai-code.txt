// FILE: src\core\composite.ts

import { Component } from "./component.js";

type CompositeCtor<T> = new (name: string, nodes?: ReadonlyArray<Component>) => T;

/**
 * Slot-based composition:
 * - createElement() returns a host element
 * - children are appended as light-DOM so <slot> renders them
 */
export abstract class Composite implements Component {
    constructor(
        public readonly name: string,
        private readonly nodes: ReadonlyArray<Component> = []
    ) {}

    get children(): ReadonlyArray<Component> {
        return this.nodes;
    }

    add(child: Component): this {
        const Ctor = this.constructor as unknown as CompositeCtor<this>;
        return new Ctor(this.name, [...this.nodes, child]);
    }

    /** Subclasses only decide the host tag (and optionally set initial props). */
    protected abstract createHostElement(): HTMLElement;

    createElement(): HTMLElement {
        const host = this.createHostElement();

        for (const child of this.children) {
            host.appendChild(child.createElement());
        }

        return host;
    }
}

// FILE: src\core\ui-element.ts

import { LitElement } from "lit";
import { property } from "lit/decorators.js";
import type { Component } from "./component.js";

export abstract class UIElement<T extends Component = Component> extends LitElement {
  // Avoid class-field shadowing: use accessor for reactive property.
  @property({ attribute: false })
  public accessor model!: T;

  protected renderChildren() {
    return this.model.children.map((child) => {
      const el = child.createElement();
      (el as any).model = child;
      return el;
    });
  }
}

# UI Framework â€” Composite + Lit 3 + Shadow DOM

This project demonstrates a fully SOLID, DRY, Open/Closed UI architecture
based on:

- Composite Pattern (OOP)
- Runtime rendering with Web Components
- Lit 3 with Shadow DOM
- Behavior Wrappers
- Layout Primitives
- Input Primitives

## Running

Serve the demo folder using any static server:

\\\
cd ui-framework/demo
npx serve .
\\\

Open:

\\\
http://localhost:3000
\\\

## Architecture

- UI Element Tree is built from pure TypeScript models
- Web Components render the tree
- Input, layout, and behavior primitives compose arbitrarily
- Token-based design system in \styles/style.css\

See diagram in \diagrams/classes.puml\.

import { html, css } from "lit";
import { customElement, property } from "lit/decorators.js";
import { UIElement } from "../core/ui-element.js";
import type { ButtonVariant } from "../behavior/button.js";

@customElement("ui-button")
export class UIButtonElement extends UIElement {
  @property({ type: String, reflect: true })
  accessor variant: ButtonVariant = "primary";

  @property({ type: Boolean, reflect: true })
  accessor disabled = false;

  static styles = css`
    :host {
      display: inline-block;
    }

    button {
      all: unset;
      box-sizing: border-box;
      display: inline-flex;
      align-items: center;
      justify-content: center;
      gap: 8px;

      padding: 8px 14px;
      border-radius: 8px;

      font: inherit;
      line-height: 1;
      cursor: pointer;
      user-select: none;

      border: 1px solid var(--ui-border-strong, #aaa);
      transition: filter 0.1s ease-in-out, transform 0.02s ease-in-out;
    }

    button:active {
      transform: translateY(1px);
    }

    button:focus-visible {
      outline: 2px solid var(--ui-primary, #0d6efd);
      outline-offset: 2px;
    }

    :host([disabled]) button {
      cursor: not-allowed;
      opacity: 0.6;
    }

    /* Variants */
    :host([variant="primary"]) button {
      background: var(--ui-primary, #0d6efd);
      color: white;
      border-color: transparent;
    }

    :host([variant="secondary"]) button {
      background: var(--ui-bg-elevated, #f7f7f7);
      color: var(--ui-text-1, #222);
    }

    :host([variant="ghost"]) button {
      background: transparent;
      color: var(--ui-text-1, #222);
      border-color: transparent;
    }

    :host([variant="ghost"]) button:hover {
      filter: brightness(0.95);
    }
  `;

  @property({ type: String, reflect: true })
  accessor label = "";

  private emitClick(e: MouseEvent) {
    if (this.disabled) return;

    this.dispatchEvent(
      new CustomEvent("ui-click", {
        detail: { label: this.label, sourceEvent: e },
        bubbles: true,
        composed: true,
      })
    );
  }

  private onKeyDown(e: KeyboardEvent) {
    if (this.disabled) return;

    if (e.key === "Enter" || e.key === " ") {
      e.preventDefault();
      this.dispatchEvent(
        new CustomEvent("ui-click", {
          detail: { label: this.label, sourceEvent: e },
          bubbles: true,
          composed: true,
        })
      );
    }
  }

  render() {
    return html`
      <button
        type="button"
        ?disabled=${this.disabled}
        @click=${(e: MouseEvent) => this.emitClick(e)}
        @keydown=${(e: KeyboardEvent) => this.onKeyDown(e)}
      >
        <slot></slot>
      </button>
    `;
  }
}

declare global {
  interface HTMLElementTagNameMap {
    "ui-button": UIButtonElement;
  }
}

// FILE: tests\fixtures\componentFixture.ts

import { expect } from "vitest";
import type { Component } from "../../src/core/component.js";

export type Fixture<TModel extends Component = Component> = Readonly<{
  model: TModel;
  element: HTMLElement;
}>;

export function createComponentFixture<TModel extends Component>(model: TModel): Fixture<TModel> {
  const element = model.createElement();
  return { model, element };
}

/** Strict DOM query: fails fast with a good error. */
export function $<T extends Element = Element>(
  root: ParentNode,
  selector: string
): T {
  const el = root.querySelector(selector);
  expect(el, `Expected element for selector: ${selector}`).not.toBeNull();
  return el as T;
}

/** Strict DOM queryAll: convenient + fails fast when empty. */
export function $all<T extends Element = Element>(
  root: ParentNode,
  selector: string
): ReadonlyArray<T> {
  const els = Array.from(root.querySelectorAll(selector)) as T[];
  expect(els.length, `Expected at least 1 element for selector: ${selector}`).toBeGreaterThan(0);
  return els;
}

/** Assert host tag name in a stable way. */
export function expectTag(el: Element, tag: string): void {
  expect(el.tagName.toLowerCase()).toBe(tag.toLowerCase());
}

/**
 * Host-shape assertion helper (solves: "possibly undefined" for host props).
 * Use it only when your model sets host properties (e.g. options arrays, state flags).
 */
export function assertHostProps<T extends object>(
  el: HTMLElement,
  props: ReadonlyArray<keyof T>
): asserts el is HTMLElement & T {
  const anyEl = el as any;
  for (const p of props) {
    expect(p in anyEl, `Expected host property "${String(p)}"`).toBe(true);
  }
}

// FILE: tests\input\button.test.ts

import { describe, it, expect } from "vitest";

import { ButtonUI } from "../../src/input/button.js";
import { LabelUI } from "../../src/input/label.js";
import { createComponentFixture, expectTag, assertHostProps, $ } from "../fixtures/componentFixture.js";

type ButtonVariant = "primary" | "secondary" | "ghost";
type ButtonHost = { variant: ButtonVariant; disabled: boolean };

function createButton(options?: ConstructorParameters<typeof ButtonUI>[1]) {
  const model = new ButtonUI(new LabelUI("Submit"), options);
  const fx = createComponentFixture(model);

  // host contract
  expectTag(fx.element, "ui-button");

  // host props contract (and TS narrowing)
  assertHostProps<ButtonHost>(fx.element, ["variant", "disabled"]);
  const host = fx.element as HTMLElement & ButtonHost;

  return { ...fx, host };
}

describe("ButtonUI (model)", () => {
  it("defaults", () => {
    const { host } = createButton();

    expect(host.variant).toBe("primary");
    expect(host.disabled).toBe(false);
  });

  it("applies options", () => {
    const { host, element } = createButton({ variant: "secondary", disabled: true });

    expect(host.variant).toBe("secondary");
    expect(host.disabled).toBe(true);
    expect(element.getAttribute("aria-disabled")).toBe("true");
  });

  it("renders child as light DOM", () => {
    const { element } = createButton();

    const label = $<HTMLElement>(element, "ui-label");
    expect(label.textContent).toBe("Submit");
  });
});
